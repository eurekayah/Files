                // 從 1 到 1000 統計數字 8 出現次數
                function countDigit8(n) {
                let total = 0;
                for (let i = 1; i <= n; i++) {
                const str = i.toString();
                for (const ch of str) {
                if (ch === '8') total++;
                }
                }
                return total;
                }

                // 執行驗證
                const result = countDigit8(1000);
